=== 旧製品 RLS セキュリティ修正 SQL ===
Supabase Dashboard → SQL Editor → 以下を全部ペースト → Run
（5製品まとめて1回で実行できます）

-- ============ FudoText ============
DROP POLICY IF EXISTS "anon_all" ON fudotext_trials;
DROP POLICY IF EXISTS "anon_insert" ON fudotext_trials;
DROP POLICY IF EXISTS "anon_select" ON fudotext_trials;
CREATE POLICY "anon_insert" ON fudotext_trials FOR INSERT TO anon WITH CHECK (true);
CREATE POLICY "anon_select" ON fudotext_trials FOR SELECT TO anon USING (true);
